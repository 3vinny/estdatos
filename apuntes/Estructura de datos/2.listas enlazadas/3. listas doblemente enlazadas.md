```c
List Delete(ElementType X, List L) { 
	Position P; 
	P = Find(X, L); 
	
	if (P != NULL) { 
		if (P->Next != NULL) 
			P->Next->Prev = P->Prev; 
		if (P->Prev != NULL) 
			P->Prev->Next = P->Next; 
		free(P); 
	} 
	return L; 
}
```
